Query_6_JOIN
====================================================================================================

Description
Top 10 highest-rated books per category

SQL Statement
SELECT
                title,
                category_name,
                rating,
                price_inr,
                in_stock
            FROM
            (
                SELECT
                    b.title,
                    c.category_name,
                    b.rating,
                    b.price_inr,
                    b.in_stock,
                    ROW_NUMBER() OVER
                    (
                        PARTITION BY c.category_name
                        ORDER BY
                            b.rating DESC,
                            b.price_inr DESC
                    ) AS row_num

                FROM books b

                INNER JOIN categories c
                    ON b.category_id = c.category_id
            )

            WHERE row_num <= 10

            ORDER BY
                category_name,
                rating DESC,
                price_inr DESC;
